# Protocol
MODEL_KEY = "qwen"
TEMPERATURE = 0.7
TOP_P = 0.9
MAX_K = 10
VOTE_KS = [2, 5, 10]
SEED = 42
LETTERS = ["A", "B", "C", "D"]
TASKS = [
    "Art_Style", "Counting", "Forensic_Detection",
    "Relative_Depth", "Spatial_Relation", "Visual_Similarity",
]


def set_seed(seed):
    torch.manual_seed(seed)
    np.random.seed(seed)
    random.seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)


def softmax_topp(logits, temperature, top_p):
    """Standard top-p + temperature softmax on a logit vector."""
    scaled = (logits / temperature).float()
    probs = torch.softmax(scaled, dim=-1)
    sorted_probs, sorted_idx = torch.sort(probs, descending=True)
    cdf = sorted_probs.cumsum(dim=-1)
    cutoff = cdf > top_p
    cutoff[..., 0] = False  # keep top-1 always
    sorted_probs[cutoff] = 0
    if sorted_probs.sum() == 0:
        sorted_probs = torch.softmax(scaled, dim=-1)
        sorted_idx = torch.arange(len(scaled), device=logits.device)
    sorted_probs = sorted_probs / sorted_probs.sum()
    return sorted_probs, sorted_idx


def sample_letter(logits, letter_token_ids, mode, gen):
    """Sample a letter from {A,B,C,D} given last-token logits.

    Three modes:
      'greedy' : argmax over letter tokens
      'stoch'  : multinomial sample after temperature+top_p on letter subspace
    """
    letter_logits = torch.tensor(
        [logits[letter_token_ids[L]].item() for L in LETTERS],
        device="cpu",
    )
    if mode == "greedy":
        return LETTERS[int(letter_logits.argmax())]
    if mode == "stoch":
        # Plain temperature softmax over the 4 letters, NO top_p. top_p=0.9
        # mathematically truncates a 2-/4-way subspace to top-1 (any positive
        # gap on 2 letters -> [1.0, 0.0]), which zeroed sampling diversity on
        # the binary tasks. The paper's self-consistency uses no top_p.
        probs = torch.softmax(letter_logits / TEMPERATURE, dim=-1)
        choice = torch.multinomial(probs, num_samples=1, generator=gen).item()
        return LETTERS[int(choice)]
    raise ValueError(mode)


def majority_with_random_tiebreak(letters, gen):
    counts = Counter(letters)
    top = counts.most_common()
    top_count = top[0][1]
    tied = [L for L, c in top if c == top_count]
    if len(tied) == 1:
        return tied[0]
    # Uniform random tie-break
    idx = torch.randint(0, len(tied), (1,), generator=gen).item()
    return tied[idx]


def evaluate_task(model, processor, task_name, samples, voting_mode,
                  out_jsonl, log):
    cfg = get_config(MODEL_KEY)
    device = next(model.parameters()).device

    # Letter token IDs once per run -- MUST match the paper harness, which
    # scores BARE-letter tokens {A:32,B:33,C:34,D:35} for Qwen2.5-VL, NOT
    # space-prefixed {362,425,356,422}. The Qwen chat template ends with
    # 'assistant\n' so the answer token follows a newline (no leading space).
    # Keep all 4 letters (paper defaults nc=4 for BLINK; do NOT restrict to
    # n_choices, or the stochastic softmax denominator diverges from the paper).
    letter_token_ids = get_choice_token_ids(processor, LETTERS)

    set_seed(SEED)
    cpu_gen = torch.Generator(device="cpu")
    cpu_gen.manual_seed(SEED)

    per_sample = []
    correct = {f"vote@{k}": 0 for k in VOTE_KS}
    correct["greedy"] = 0
    total = 0

    for s_idx, sample in enumerate(samples):
        gt = parse_mc_answer(sample["answer"])
        if gt is None:
            continue
        try:
            inputs, _ = prepare_inputs(
                MODEL_KEY, processor, sample["prompt"], sample["images"], device,
            )
        except Exception:
            continue

        # Baseline greedy forward (sample 1, deterministic)
        with torch.no_grad():
            out_n = model(**inputs)
        logits_n = out_n.logits[0, -1, :].float()
        greedy_letter = sample_letter(logits_n, letter_token_ids, "greedy", cpu_gen)

        # Sample k-1 more
        all_letters = [greedy_letter]  # sample 1 = greedy in mixed mode
        if voting_mode == "greedy":
            # All k samples = greedy = identical -> trivially same
            for _ in range(MAX_K - 1):
                all_letters.append(greedy_letter)
        elif voting_mode in ("mixed", "stoch"):
            if voting_mode == "stoch":
                # Replace sample 1 with a stochastic sample too
                all_letters[0] = sample_letter(
                    logits_n, letter_token_ids, "stoch", cpu_gen)
            for _ in range(MAX_K - 1):
                # NOTE: we re-use the same baseline forward's logits
                # because they are deterministic; only the SAMPLING is
                # stochastic. This matches the paper's
                # final_experiments_mar28.py implementation (samples 2..k
                # all draw from the same logits_n via multinomial).
                #
                # If we wanted "true" self-consistency we would re-forward
                # the model with do_sample=True, but BLINK MCQA-format
                # makes that ill-defined (single-token answer = no
                # sampling chain). The all-from-one-logits approach is
                # standard for MCQA self-consistency.
                lab = sample_letter(logits_n, letter_token_ids,
                                    "stoch", cpu_gen)
                all_letters.append(lab)
        else:
            raise ValueError(voting_mode)

        # Compute vote outcomes for each k
        votes_at_k = {}
        for k in VOTE_KS:
            letters_k = all_letters[:k]
            # Match the original Table 3 / exp6 tie-break EXACTLY: Counter.most_common
            # breaks ties by first-encountered order, and all_letters[0] is the greedy
            # sample in mixed mode, so ties resolve toward greedy (as described to the
            # The earlier random tie-break deviated from the reported table.
            votes_at_k[f"vote@{k}"] = Counter(letters_k).most_common(1)[0][0]

        # Greedy-only "sample 1" prediction (not the same as voting@1)
        greedy_pred = all_letters[0] if voting_mode == "stoch" else greedy_letter

        total += 1
        correct["greedy"] += int(greedy_letter == gt)
        for k_str, pred in votes_at_k.items():
            correct[k_str] += int(pred == gt)

        per_sample.append({
            "idx": sample.get("idx", s_idx),
            "gt": gt,
            "greedy": greedy_letter,
            "all_letters": all_letters,
            "votes": votes_at_k,
        })
        del out_n
        if total % 25 == 0:
            torch.cuda.empty_cache()

    with open(out_jsonl, "w") as f:
        for r in per_sample:
            f.write(json.dumps(r) + "\n")

    n = max(1, total)
    return {
        "task": task_name,
        "voting_mode": voting_mode,
        "n": total,
        "greedy_acc": correct["greedy"] / n,
        **{f"vote@{k}_acc": correct[f"vote@{k}"] / n for k in VOTE_KS},
        **{f"vote@{k}_delta": (correct[f"vote@{k}"] - correct["greedy"]) / n
           for k in VOTE_KS},
    }


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--sanity", action="store_true")
    ap.add_argument("--voting_modes", nargs="+",
                    default=["mixed", "stoch", "greedy"])
    ap.add_argument("--tasks", nargs="+", default=TASKS)
    args = ap.parse_args()

    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    out_dir = REPO_ROOT / "results" / f"rebuttal_vote_kclean_{ts}"
    out_dir.mkdir(parents=True, exist_ok=True)
    (out_dir / "per_sample").mkdir(exist_ok=True)
    log_path = out_dir / "vote_kclean_log.txt"
    log_file = open(log_path, "a", buffering=1)

    def log(m):
        print(m)
        log_file.write(m + "\n")

    log(f"=== rebuttal_vote_kclean ({ts}) ===")
    log(f"voting_modes={args.voting_modes}  tasks={args.tasks}")
    log(f"temperature={TEMPERATURE} top_p={TOP_P} seed={SEED} max_k={MAX_K}")

    cfg = get_config(MODEL_KEY)
    model, processor, _ = load_model(MODEL_KEY)
    log(f"Loaded {MODEL_KEY}")

    max_per_task = 8 if args.sanity else None
    blink_data = load_blink(tasks=args.tasks, split="val",
                            max_per_task=max_per_task)

    summary = {
        "config": {
            "model": MODEL_KEY, "temperature": TEMPERATURE,
            "top_p": TOP_P, "seed": SEED, "max_k": MAX_K,
            "vote_ks": VOTE_KS, "sanity": args.sanity,
        },
        "results": [],
        "timestamp": ts,
    }

    for mode in args.voting_modes:
        log(f"\n── voting_mode={mode} ──")
        for task_name in args.tasks:
            samples = blink_data.get(task_name, [])
            if not samples:
                continue
            out_jsonl = out_dir / "per_sample" / f"{task_name}_{mode}.jsonl"
            log(f"  {task_name}_{mode}: n={len(samples)}")
            t0 = time.time()
            r = evaluate_task(model, processor, task_name, samples, mode,
                              str(out_jsonl), log)
            r["wallclock_min"] = (time.time() - t0) / 60.0
            summary["results"].append(r)
            log(f"    greedy={r['greedy_acc']:.4f}  "
                f"vote@2={r['vote@2_acc']:.4f} (Δ={r['vote@2_delta']:+.4f})  "
                f"vote@5={r['vote@5_acc']:.4f} (Δ={r['vote@5_delta']:+.4f})  "
                f"vote@10={r['vote@10_acc']:.4f} (Δ={r['vote@10_delta']:+.4f})")
            with open(out_dir / "vote_kclean_summary.json", "w") as f:
                json.dump(summary, f, indent=2)
            gc.collect()
            torch.cuda.empty_cache()

    readout = out_dir / "vote_kclean_readout.txt"
    with open(readout, "w") as f:
        f.write(f"Vote@k clean re-run summary ({ts})\n")
        f.write(f"Model: {MODEL_KEY}  Temp: {TEMPERATURE}  "
                f"Top-p: {TOP_P}  Seed: {SEED}\n\n")
        f.write(f"{'Task':<22s}{'Mode':<8s}"
                f"{'Greedy':>10s}{'V@2 Δ':>10s}{'V@5 Δ':>10s}{'V@10 Δ':>10s}\n")
        f.write("-" * 70 + "\n")
        for r in summary["results"]:
            f.write(f"{r['task']:<22s}{r['voting_mode']:<8s}"
                    f"{r['greedy_acc']:9.4f} "
                    f"{r['vote@2_delta']:+9.4f} "
                    f"{r['vote@5_delta']:+9.4f} "
                    f"{r['vote@10_delta']:+9.4f}\n")
    log(f"\nDone. Readout: {readout}")
    log_file.close()


if __name__ == "__main__":
    main()
